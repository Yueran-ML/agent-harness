# Setup

1. Copy `config/settings.ini` to `config/local.ini`.
2. Run `python3 src/app.py`.

[app]

===== FILE: src/app.py =====
import configparser


def main():
    config = configparser.ConfigParser()
    config.read("config/settings.ini")
    print("Hello from", config["app"]["name"])


if __name__ == "__main__":
    main()
